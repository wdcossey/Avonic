// Measures ion-refresher's md native path in Chromium: pull frames, release, refreshing, complete.
// node measure.js <repoRoot> <outDir>
const path = require('path');
const fs = require('fs');
const http = require('http');
const ROOT = path.resolve(process.argv[2]);
const OUT = path.resolve(process.argv[3]);
const { chromium } = require(path.join(ROOT, 'node_modules', 'playwright-core'));
const roboto = require(path.join(ROOT, 'tools', 'fonts', 'roboto', 'roboto'));

const items = Array.from({ length: 20 }, (_, i) => `<ion-item><ion-label>Item ${i + 1}</ion-label></ion-item>`).join('');
const header = process.argv[4] !== 'noheader';
const snippet = `<ion-app>${header ? '<ion-header><ion-toolbar><ion-title>Refresher</ion-title></ion-toolbar></ion-header>' : ''}<ion-content><ion-refresher slot="fixed"><ion-refresher-content></ion-refresher-content></ion-refresher><ion-list>${items}</ion-list></ion-content></ion-app>`;
const MIME = { '.js': 'text/javascript', '.mjs': 'text/javascript', '.css': 'text/css', '.html': 'text/html', '.json': 'application/json', '.svg': 'image/svg+xml', ...roboto.MIME };
const pageHtml = `<!doctype html><html mode="md"><head><meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1, viewport-fit=cover">
<link rel="stylesheet" href="/node_modules/@ionic/core/css/ionic.bundle.css">${roboto.LINK}
<script>window.Ionic = { config: { mode: 'md' } };</script>
<script type="module" src="/node_modules/@ionic/core/dist/ionic/ionic.esm.js"></script>
<style>html,body{margin:0;height:100%;background:var(--ion-background-color,#fff)} *{user-select:none;-webkit-user-select:none}</style>
</head><body>${snippet}</body></html>`;
const server = http.createServer((req, res) => {
    const url = decodeURIComponent(req.url.split('?')[0]);
    if (url === '/') { res.writeHead(200, { 'content-type': 'text/html' }); return res.end(pageHtml); }
    const file = path.join(ROOT, url);
    if (!file.startsWith(ROOT) || !fs.existsSync(file) || fs.statSync(file).isDirectory()) { res.writeHead(404); return res.end(); }
    res.writeHead(200, { 'content-type': MIME[path.extname(file)] ?? 'application/octet-stream' });
    fs.createReadStream(file).pipe(res);
});

(async () => {
    await new Promise(r => server.listen(0, '127.0.0.1', r));
    fs.mkdirSync(OUT, { recursive: true });
    const browser = await chromium.launch();
    const ctx = await browser.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 2 });
    const page = await ctx.newPage();
    const errors = [];
    page.on('pageerror', e => errors.push(String(e)));
    await page.goto(`http://127.0.0.1:${server.address().port}/`);
    await page.waitForFunction(() => document.querySelector('ion-refresher')?.classList.contains('hydrated') && document.querySelector('ion-refresher-content')?.classList.contains('hydrated'));
    await roboto.loadRoboto(page);
    await page.waitForTimeout(800);
    await page.evaluate(() => {
        window.__log = [];
        const r = document.querySelector('ion-refresher');
        for (const n of ['ionPull', 'ionPullStart', 'ionPullEnd', 'ionStart', 'ionRefresh'])
            r.addEventListener(n, e => window.__log.push({ n, t: performance.now(), d: e.detail ? JSON.stringify(e.detail.reason ?? 'complete-fn') : null }));
    });
    const probe = async label => page.evaluate(label => {
        const r = document.querySelector('ion-refresher');
        const q = s => r.querySelector(s);
        const rect = e => { if (!e) return null; const b = e.getBoundingClientRect(); return [b.x, b.y, b.width, b.height].map(v => Math.round(v * 100) / 100); };
        const cs = (e, p) => e ? getComputedStyle(e).getPropertyValue(p) : null;
        const pi = q('.refresher-pulling-icon'), sp = q('.refresher-pulling ion-spinner');
        const circle = sp?.shadowRoot?.querySelector('circle');
        const ri = q('.refresher-refreshing-icon'), rsp = q('.refresher-refreshing ion-spinner');
        return {
            label, t: performance.now(), classes: r.className, refresher: rect(r),
            pulling: { display: cs(q('.refresher-pulling'), 'display'), rect: rect(pi), transform: cs(pi, 'transform'),
                spinner: cs(sp, 'transform'), dash: cs(circle, 'stroke-dasharray'), dashoffset: cs(circle, 'stroke-dashoffset'),
                sac: cs(q('.spinner-arrow-container'), 'opacity'), arrowContainer: cs(q('.arrow-container'), 'transform'),
                arrow: cs(q('.arrow-container ion-icon'), 'transform'), arrowRect: rect(q('.arrow-container ion-icon')),
                bg: cs(pi, 'background-color'), border: cs(pi, 'border'), shadow: cs(pi, 'box-shadow'), radius: cs(pi, 'border-radius'),
                color: cs(sp, 'color'), spinnerRect: rect(sp) },
            refreshing: { display: cs(q('.refresher-refreshing'), 'display'), rect: rect(ri), transform: cs(ri, 'transform'), transition: cs(ri, 'transition'),
                spinnerRect: rect(rsp) },
            scrollTop: document.querySelector('ion-content').shadowRoot.querySelector('.inner-scroll').scrollTop,
        };
    }, label);
    const results = [];
    const shot = async name => { await page.screenshot({ path: path.join(OUT, name + '.png') }); };
    results.push(await probe('idle'));
    const x = 195, y0 = 200; const SHORT = process.argv[5] === "short";
    let y = y0;
    const moveTo = async target => { while (y < target) { y = Math.min(target, y + 6); await page.mouse.move(x, y); await page.waitForTimeout(16); } };
    if (!SHORT) {
    // Pull: mouse down, small steps; Ionic's gesture captures at 5 px, and progress = delta/360 from the capture point.
    await page.mouse.move(x, y0);
    await page.mouse.down();
    await moveTo(y0 + 8); await page.waitForTimeout(50);
    const cap = y; // approximate capture point
    for (const p of [0.1, 0.25, 0.4, 0.5]) {
        await moveTo(Math.round(cap + 360 * p)); await page.waitForTimeout(60);
        results.push(await probe('pull-' + p)); await shot('pull-' + p);
    }
    await page.mouse.up();
    const t0 = await page.evaluate(() => performance.now());
    for (const ms of [30, 70, 110, 160, 300]) {
        const now = await page.evaluate(() => performance.now()); if (t0 + ms > now) await page.waitForTimeout(t0 + ms - now);
        results.push(await probe('release-' + ms)); if (ms === 300) await shot('refreshing');
    }
    // Complete
    await page.evaluate(() => document.querySelector('ion-refresher').complete());
    const t1 = await page.evaluate(() => performance.now());
    for (const ms of [20, 60, 120, 200, 260, 400]) {
        const now = await page.evaluate(() => performance.now()); if (t1 + ms > now) await page.waitForTimeout(t1 + ms - now);
        results.push(await probe('complete-' + ms)); if (ms === 120) await shot('completing');
    }
    }
    // Short pull, cancelled
    await page.waitForTimeout(300);
    y = y0; await page.mouse.move(x, y0); await page.mouse.down();
    await moveTo(y0 + 8); await page.waitForTimeout(50);
    const cap2 = y;
    await moveTo(Math.round(cap2 + 360 * 0.3)); await page.waitForTimeout(60);
    results.push(await probe('short-0.3'));
    await page.mouse.up();
    const t2 = await page.evaluate(() => performance.now());
    for (const ms of [40, 100, 160, 260]) {
        const now = await page.evaluate(() => performance.now()); if (t2 + ms > now) await page.waitForTimeout(t2 + ms - now);
        results.push(await probe('short-release-' + ms));
    }
    // Upward drag at the top: does it pull?
    await page.waitForTimeout(300);
    const log = await page.evaluate(() => window.__log);
    fs.writeFileSync(path.join(OUT, 'measure.json'), JSON.stringify({ snippet, errors, results, log }, null, 1));
    console.log('done', results.length, 'probes', log.length, 'events', errors.length, 'errors');
    await browser.close(); server.close();
})().catch(e => { console.error(e); server.close(); process.exit(1); });
